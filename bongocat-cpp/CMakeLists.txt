cmake_minimum_required(VERSION 3.16)
project(BongoCatCppDemo LANGUAGES CXX RC)
if(NOT WIN32)
    message(FATAL_ERROR "This demo supports Windows only")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(BongoCatDemo WIN32 src/main.cpp src/renderer.cpp src/settings_panel.cpp app.rc)
target_link_libraries(BongoCatDemo PRIVATE gdiplus gdi32 user32 shell32 psapi comctl32)
if(MINGW)
    target_link_options(BongoCatDemo PRIVATE -municode -static)
    target_compile_options(BongoCatDemo PRIVATE -Wall -Wextra)
elseif(MSVC)
    target_compile_options(BongoCatDemo PRIVATE /W4 /utf-8)
    set_property(TARGET BongoCatDemo PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
add_custom_command(TARGET BongoCatDemo POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E make_directory "$<TARGET_FILE_DIR:BongoCatDemo>/assets"
    COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CMAKE_SOURCE_DIR}/assets/cat-atlas-v3.png" "$<TARGET_FILE_DIR:BongoCatDemo>/assets/cat-atlas-v3.png")
enable_testing()
add_executable(animation_test tests/animation_test.cpp)
add_test(NAME animation COMMAND animation_test)
add_executable(input_test tests/input_test.cpp)
add_test(NAME input COMMAND input_test)
